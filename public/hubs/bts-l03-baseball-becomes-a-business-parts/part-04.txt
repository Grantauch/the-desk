function upscaleAndSharpen(img){
  if(img.dataset.hdProcessed==='1')return;
  img.dataset.hdProcessed='1';
  try{
    const scale=4,w=img.naturalWidth*scale,h=img.naturalHeight*scale;
    if(!img.naturalWidth||!img.naturalHeight)return;
    const c=document.createElement('canvas');c.width=w;c.height=h;
    const x=c.getContext('2d',{willReadFrequently:true});
    x.imageSmoothingEnabled=true;x.imageSmoothingQuality='high';
    x.drawImage(img,0,0,w,h);
    const d=x.getImageData(0,0,w,h),p=d.data,copy=new Uint8ClampedArray(p);
    const amount=.45;
    for(let yy=1;yy<h-1;yy++)for(let xx=1;xx<w-1;xx++){
      const i=(yy*w+xx)*4;
      for(let k=0;k<3;k++){
        const center=copy[i+k],
          blur=(copy[i-4+k]+copy[i+4+k]+copy[i-w*4+k]+copy[i+w*4+k]+center*4)/8;
        p[i+k]=Math.max(0,Math.min(255,center+(center-blur)*amount));
      }
    }
    x.putImageData(d,0,0);
    img.src=c.toDataURL('image/webp',.9);
  }catch(e){console.warn('StoryHub image enhancement skipped',e)}
}
(async()=>{for(const img of document.querySelectorAll('img[data-b64]')){try{const r=await fetch(img.dataset.b64,{cache:'no-store'});if(!r.ok)throw new Error(r.status);const b=(await r.text()).trim();img.onload=()=>upscaleAndSharpen(img);img.src='data:image/webp;base64,'+b;}catch(e){console.error('StoryHub image load failed',img.dataset.b64,e);}}})();
